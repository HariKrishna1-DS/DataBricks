Different ways on Reading data from delta table

Step 1: Create Delta Table

----------------------------------
from delta.tables import *

DeltaTable.create(spark) \
     .tableName("dev.demodb.emp4") \
     .addColumn("eid","INT") \
     .addColumn("ename","STRING") \
     .addColumn("sal","DOUBLE") \
     .addColumn("sex","STRING") \
    .addColumn("dno","INT") \
    .execute()
--------------------------------------

Step 2: inserting data(sql style)

---------------------------------------------
%sql
insert into dev.demodb.emp4 values (101,"Raj",50000,"M",11),(102,"Ravi",60000,"M",12),(103,"Suresh",70000,"M",13);
---------------------------------------------
=========================================================================
Step 3: Retrieving data

----------------------------
%sql
select * from dev.demodb.emp4;
