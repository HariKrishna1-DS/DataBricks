4. #II-method :Pyspark style ----->using Table instance
from pyspark.sql.functions import *
from delta.tables import *
deltatable_instance=DeltaTable.forName(spark,"dev1.demodb.emp")
deltatable_instance.update(condition="ename='Miller'",set={"sal":"75000"})

spark.sql("select * from dev1.demodb.emp").display()
